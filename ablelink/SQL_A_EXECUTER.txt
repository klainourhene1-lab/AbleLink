-- Créer la base de données
CREATE DATABASE IF NOT EXISTS ablelink_db;
USE ablelink_db;

-- Table pour les Success Stories
CREATE TABLE IF NOT EXISTS success_stories (
    id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(255) NOT NULL,
    author VARCHAR(255) NOT NULL,
    category VARCHAR(100) NOT NULL,
    description TEXT NOT NULL,
    image VARCHAR(255),
    likes INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- Table pour les Commentaires
CREATE TABLE IF NOT EXISTS comments (
    id INT PRIMARY KEY AUTO_INCREMENT,
    story_id INT NOT NULL,
    author VARCHAR(255) NOT NULL,
    comment_text TEXT NOT NULL,
    likes INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (story_id) REFERENCES success_stories(id) ON DELETE CASCADE
);

1. CREATE DATABASE IF NOT EXISTS ablelink_db;
   → Crée la base de données nommée "ablelink_db"
   → "IF NOT EXISTS" = si elle n'existe pas déjà

2. USE ablelink_db;
   → Utilise la base ablelink_db pour les commandes suivantes

3. CREATE TABLE IF NOT EXISTS success_stories (...)
   → Crée la table "success_stories" avec ces colonnes:
      • id: Numéro unique (clé primaire, auto-incrémenté)
      • title: Titre de l'histoire (255 caractères max)
      • author: Nom de l'auteur (255 caractères max)
      • category: Catégorie (Career Growth, Technology, etc)
      • description: Description complète (texte long)
      • image: URL de l'image
      • likes: Nombre de likes (commence à 0)
      • created_at: Date de création (auto)
      • updated_at: Date de modification (auto)

4. CREATE TABLE IF NOT EXISTS comments (...)
   → Crée la table "comments" avec ces colonnes:
      • id: Numéro unique
      • story_id: Référence à une story (FK = Foreign Key)
      • author: Nom du commentateur
      • comment_text: Texte du commentaire
      • likes: Nombre de likes
      • created_at: Date de création
      • FOREIGN KEY: Lien avec success_stories (suppression en cascade)

5. INSERT INTO success_stories...
   → Ajoute 2 histoires d'exemple

6. INSERT INTO comments...
   → Ajoute 2 commentaires sur la première histoire
